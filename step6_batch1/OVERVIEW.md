## Context: What Step 6 Is

After `forge execute` AI-implements all workstreams, `forge integrate` verifies the whole system works together by:
1. Reading `execute.json` (what was executed), `plan.json` (what was supposed to be built), `verify.json` (safety constraints)
2. Building an AI prompt asking for integration tests covering the whole
3. The AI generates test files
4. Test runner executes them against the actual codebase
5. Producing `integrate.json` + `integration-report.md`

Step 5 operates on **one workstream at a time**. Step 6 operates on the **entire codebase** — all workstreams together.

**Key constraint:** Do NOT create a new AI model connector. Reuse `src/execute/model-connector.ts`.

```typescript
import path from "node:path";
import type {
  ExecuteArtifact,

  VerifyArtifact,
  IntegrateArtifact,
  TestRunResult,

} from "./types.js";

export function buildIntegrateArtifact(
  executeArtifact: ExecuteArtifact,
  planArtifact: PlanArtifact | null,
  verifyArtifact: VerifyArtifact | null,
  testResult: TestRunResult,
  aiModelUsed: string,
  aiConfig: AIModelInfo | undefined,
  schemaVersion: string,
  forgeVersion: string,
): IntegrateArtifact {
  const now = new Date().toISOString();
  
  const goal = planArtifact?.task?.goal 
    ?? planArtifact?.goal 
    ?? "Unknown goal";

  const totalChangesMade = executeArtifact.workstreams
    .reduce((sum, ws) => {
      if (ws.aiChangesCount !== undefined) return sum + ws.aiChangesCount;
      return sum + (ws.changesMade?.length ?? 0);
    }, 0);

  const workstreamsSummary = {
    total: executeArtifact.workstreams.length,
    completed: executeArtifact.workstreams.filter(ws => ws.state === "completed").length,
    failed: executeArtifact.workstreams.filter(ws => ws.state === "failed").length,
    totalChangesMade,
  };

  const recommendations = testResult.tests
    .filter(t => t.status === "failed" && t.recommendation)
    .map(t => t.recommendation!);

  return validateIntegrateArtifact({
    schemaVersion,
    forgeVersion,
    createdAt: now,
    executeSource: executeArtifact.splitSource ?? ".forge/split.json",
    planSource: planArtifact ? ".forge/plan.json" : "",
    verifySource: verifyArtifact ? ".forge/verify.json" : "",
    goal,
    workstreamsSummary,
    tests: testResult.tests,
    testFiles: testResult.testFiles,
    summary: {
      total: testResult.tests.length,
      passed: testResult.tests.filter(t => t.status === "passed").length,
      failed: testResult.tests.filter(t => t.status === "failed").length,
      skipped: testResult.tests.filter(t => t.status === "skipped").length,
      durationMs: testResult.durationMs,
      testFilesGenerated: testResult.testFiles.length,
      aiModelUsed,
    },
    aiConfig,
    recommendations,
  });
}
```

### 4. `src/integrate/prompt-builder.ts` — AI prompt builder

```typescript
import { promises as fs } from "fs";
import path from "node:path";
import crypto from "node:crypto";
import type {
  ExecuteArtifact,
  PlanArtifact,
  VerifyArtifact,
  PromptBuildContext,
  BuiltPrompt,
} from "./types.js";

export async function detectTestFramework(
  repoRoot: string
): Promise<{ name: string; language: string; testCommand: string }> {
  // Try package.json
  try {
    const pkg = JSON.parse(
      await fs.readFile(path.join(repoRoot, "package.json"), "utf-8")
    );
    const testScript: string = pkg.scripts?.test ?? "";
    
    if (testScript.includes("jest") || pkg.dependencies?.jest || pkg.devDependencies?.jest) {
      return { name: "jest", language: "typescript", testCommand: "npm test -- --passWithNoTests" };
    }
    if (testScript.includes("vitest") || pkg.devDependencies?.vitest) {
      return { name: "vitest", language: "typescript", testCommand: "npm test -- --passWithNoTests" };
    }
    if (testScript.includes("mocha")) {
      return { name: "mocha", language: "typescript", testCommand: "npm test" };
    }
    if (testScript.includes("pytest") || testScript.includes("python -m pytest")) {
      return { name: "pytest", language: "python", testCommand: "python -m pytest" };
    }
    if (testScript) {
      return { name: "npm", language: "unknown", testCommand: "npm test" };
    }
  } catch { /* package.json not found */ }

  // Try pytest
  try {
    await fs.access(path.join(repoRoot, "pytest.ini"));
    return { name: "pytest", language: "python", testCommand: "python -m pytest" };
  } catch { /* no pytest.ini */ }

  try {
    const pyproject = await fs.readFile(path.join(repoRoot, "pyproject.toml"), "utf-8");
    if (pyproject.includes("[tool.pytest")) {
      return { name: "pytest", language: "python", testCommand: "python -m pytest" };
    }
  } catch { /* no pyproject.toml */ }

  // Default
  return { name: "npm", language: "unknown", testCommand: "npm test" };
}

export async function getChangedFileContents(
  executeArtifact: ExecuteArtifact,
  repoRoot: string
): Promise<Record<string, string>> {
  const contents: Record<string, string> = {};
  
  for (const ws of executeArtifact.workstreams) {
    if (!ws.changesMade) continue;
    for (const change of ws.changesMade) {
      if (change.action === "delete") continue;
      try {
        const fullPath = path.isAbsolute(change.file) 
          ? change.file 
          : path.join(repoRoot, change.file);
        contents[change.file] = await fs.readFile(fullPath, "utf-8");
      } catch {
        contents[change.file] = `[FILE NOT FOUND: ${change.file}]`;
      }
    }
  }
  
  return contents;
}

export async function buildIntegrationTestPrompt(
  ctx: PromptBuildContext
): Promise<BuiltPrompt> {
  const framework = ctx.testFramework
    ? { name: ctx.testFramework, language: "unknown", testCommand: `npm test` }
    : await detectTestFramework(ctx.repoRoot);

  // Build workstreams summary
  const workstreamsSection = ctx.executeArtifact.workstreams
    .map(ws => {
      const changes = ws.changesMade
        ?.map(c => `  - ${c.action}: ${c.file} (+${c.linesAdded}/-${c.linesRemoved})`)
        .join("\n") ?? "  (no changes recorded)";
      return `# ${ws.title} (${ws.workstreamId})\nState: ${ws.state}\nChanges:\n${changes}`;
    })
    .join("\n\n");

  // Build plan items section
  let planSection = "(no plan.json available)";
  if (ctx.planArtifact) {
    const items = ctx.planArtifact.items ?? [];
    planSection = items
      .map((item: any) => `- ${item.requirement ?? item.title}: ${item.category ?? "general"}`)
      .join("\n");
  }

  // Build verify constraints section
  let verifySection = "(no verify.json available)";
  if (ctx.verifyArtifact) {
    const constraints: string[] = [];
    if ((ctx.verifyArtifact as any).conflictZones) {
      constraints.push(...(ctx.verifyArtifact as any).conflictZones.map((z: any) => `CONFLICT ZONE: ${z.description}`));
    }
    if ((ctx.verifyArtifact as any).findings) {
      constraints.push(...(ctx.verifyArtifact as any).findings.map((f: any) => `FINDING: ${f.message}`));
    }
    if ((ctx.verifyArtifact as any).constraints) {
      constraints.push(...(ctx.verifyArtifact as any).constraints.map((c: any) => `CONSTRAINT: ${c}`));
    }
    verifySection = constraints.length > 0 ? constraints.join("\n") : "(no constraints found)";
  }

  // Build changed files section
  const changedFiles = await getChangedFileContents(ctx.executeArtifact, ctx.repoRoot);
  const filesSection = Object.entries(changedFiles)
    .map(([filePath, content]) => `FILE: ${filePath}\n---\n${content}\n---`)
    .join("\n\n");

  // Build existing tests section
  let existingTestsSection = "(no existing tests found)";
n  try {
    const testFiles: string[] = [];
    const findTests = async (dir: string): Promise<void> => {
      const entries = await fs.readdir(dir, { withFileTypes: true });
      for (const entry of entries) {
        const fullPath = path.join(dir, entry.name);
        if (entry.isDirectory() && !entry.name.startsWith(".") && entry.name !== "node_modules") {
          await findTests(fullPath);
        } else if (entry.isFile() && (entry.name.includes(".test.") || entry.name.includes(".spec.") || entry.name.endsWith(".test.ts") || entry.name.endsWith(".spec.ts") || entry.name.endsWith("_test.py"))) {
          testFiles.push(fullPath);
        }
      }
    };
    await findTests(path.join(ctx.repoRoot, "tests"));
    if (testFiles.length > 0) {
      existingTestsSection = testFiles.join("\n");
    }
  } catch { /* no tests directory */ }

  const goal = ctx.planArtifact?.task?.goal ?? ctx.planArtifact?.goal ?? "Implement the planned features";

  const prompt = `# SYSTEM ROLE
You are a skilled software engineer writing integration tests for a codebase.

# ORIGINAL TASK GOAL
${goal}

# TEST FRAMEWORK
Use: ${framework.name}
Language: ${framework.language}
Run tests with: ${framework.testCommand}

# EXECUTED WORKSTREAMS
${workstreamsSection}

# PLAN ITEMS (what was supposed to be built)
${planSection}

# VERIFY CONSTRAINTS (safety rules from verification)
${verifySection}

# CHANGED FILES (actual file contents after AI execution)
${filesSection}

# EXISTING TESTS (already covered — do not duplicate)
${existingTestsSection}

# YOUR TASK
Write integration tests that verify the whole system works together after the above changes.
Focus on:
- End-to-end workflows covering multiple files/workstreams
- Integration points between components changed by different workstreams
- Key user-facing functionality from the original goal
- DO NOT write unit tests (those should already exist or be covered elsewhere)

# OUTPUT FORMAT
Return a JSON array of test files to create:
\`\`\`json
[
  {
    "path": "tests/integration/my-test.test.ts",
    "framework": "${framework.name}",
    "language": "${framework.language}",
    "content": "import { describe, it, expect } from '@jest/globals';\n..."
  }
]
\`\`\`
Do not include any additional text — only the JSON array.`;

  const promptHash = crypto.createHash("sha256").update(prompt).digest("hex");

  return {
    prompt,
    promptHash,
    detectedFramework: framework,
  };
}
```

### 5. `src/integrate/test-runner.ts` — Execute generated tests

```typescript
import { promises as fs } from "fs";
import path from "node:path";
import { exec } from "node:child_process";
import { promisify } from "node:util";
import type { IntegrationTestFile, TestRunResult, IntegrationTestCase } from "./types.js";

const execAsync = promisify(exec);

function countTestsInContent(content: string, framework: string): number {
  if (framework === "pytest") {
    return (content.match(/def test_\w+/g) ?? []).length;
  }
  return (content.match(/\bit\s*\(/g) ?? []).length;
}

export async function runIntegrationTests(
  testFiles: IntegrationTestFile[],
  repoRoot: string,
  testCommand?: string
): Promise<TestRunResult> {
  const startTime = Date.now();
  const tests: IntegrationTestCase[] = [];
  const writtenFiles: IntegrationTestFile[] = [];

  // Write test files to disk
  for (const tf of testFiles) {
    const fullPath = path.isAbsolute(tf.path) ? tf.path : path.join(repoRoot, tf.path);
    await fs.mkdir(path.dirname(fullPath), { recursive: true });
    await fs.writeFile(fullPath, tf.content ?? "", "utf-8");
    writtenFiles.push({
      path: fullPath,
      testCount: tf.testCount ?? countTestsInContent(tf.content ?? "", tf.framework),
      language: tf.language,
      framework: tf.framework,
    });
  }

  if (writtenFiles.length === 0) {
    return {
      success: false,
      tests: [],
      testFiles: [],
      durationMs: Date.now() - startTime,
      error: "No test files were generated",
    };
  }

  // Run the test command
  const command = testCommand ?? "npm test -- --passWithNoTests";
  
  try {
    const { stdout, stderr } = await execAsync(command, {
      cwd: repoRoot,
      timeout: 300000, // 5 minute timeout
    });
    
    const output = stdout + stderr;
    const passed = parseTestOutput(output, "passed");
    const failed = parseTestOutput(output, "failed");
    
    const allTests: IntegrationTestCase[] = [
      ...passed.map((name, i) => ({
        id: `int-test-${i + 1}`,
        name,
        status: "passed" as const,
        durationMs: undefined,
      })),
      ...failed.map((item, i) => ({
        id: `int-test-fail-${i + 1}`,
        name: item.name,
        status: "failed" as const,
        error: item.error,
        recommendation: undefined,
        durationMs: undefined,
      })),
    ];

    return {
      success: failed.length === 0,
      tests: allTests,
      testFiles: writtenFiles,
      durationMs: Date.now() - startTime,
    };
  } catch (err) {
    const errorMessage = err instanceof Error ? err.message : String(err);
    
    // Try to parse partial results from error output
    if (errorMessage.includes("test") || errorMessage.includes("fail")) {
      return {
        success: false,
        tests: [{
          id: "int-test-1",
          name: "Integration test suite",
          status: "failed",
          error: errorMessage,
          durationMs: Date.now() - startTime,
        }],
        testFiles: writtenFiles,
        durationMs: Date.now() - startTime,
        error: errorMessage,
      };
    }

    return {
      success: false,
      tests: [],
      testFiles: writtenFiles,
      durationMs: Date.now() - startTime,
      error: `Test runner crashed: ${errorMessage}`,
    };
  }
}

function parseTestOutput(
  output: string,
  type: "passed" | "failed"
): Array<{ name: string; error?: string }> {
  const results: Array<{ name: string; error?: string }> = [];
  
  if (type === "passed") {
    const passMatches = output.matchAll(/✓ (.+?) \((?:\d+)ms\)/g);
    for (const match of passMatches) {
      results.push({ name: match[1] });
    }
    // Jest styled output
    const jestPass = output.matchAll(/PASS\s+(.+)/g);
    for (const match of jestPass) {
      results.push({ name: match[1].trim() });
    }
  } else {
    const failMatches = output.matchAll(/✕ (.+)/g);
    for (const match of failMatches) {
      results.push({ name: match[1] });
    }
    // Extract failure details
    const failBlocks = output.split(/FAIL\s+/).slice(1);
    for (const block of failBlocks) {
      const lines = block.split("\n");
      const testName = lines[0]?.trim() ?? "Unknown test";
      const errorLines = lines
        .filter(l => l.includes("Error:") || l.includes("expect("))
        .slice(0, 3);
      results.push({ name: testName, error: errorLines.join(" | ") });
    }
  }
  
  return results;
}
```

```typescript
export async function runIntegrateCommand(
  options: IntegrateCommandOptions = {}
): Promise<IntegrateCommandResult> {
  const repoRoot = options.repo ?? process.cwd();
  
  console.log("Welcome to Forge Integrate (V1)\n");

  // Load execute artifact (required)
  let executeArtifact: ExecuteArtifact;
  try {
    executeArtifact = await loadExecuteArtifact(repoRoot);
  } catch {
    return {
      status: "failed",
      summary: "execute.json not found. Run 'forge execute' first.",
      artifactPath: "",
      outputRoot: repoRoot,
      failure: { code: "NO_EXECUTE_ARTIFACT", message: "execute.json not found. Run 'forge execute' first." },
    };
  }

  // Check workstreams
  if (executeArtifact.workstreams.length === 0) {
    return {
      status: "failed",
      summary: "No workstreams found in execute.json.",
      artifactPath: "",
      outputRoot: repoRoot,
      failure: { code: "NO_WORKSTREAMS", message: "No workstreams found in execute.json." },
    };
  }

  const allFailed = executeArtifact.workstreams.every(ws => ws.state === "failed");
  if (allFailed) {
    return {
      status: "failed",
      summary: "All workstreams failed in execute.json. Integration is meaningless.",
      artifactPath: "",
      outputRoot: repoRoot,
      failure: { code: "ALL_WORKSTREAMS_FAILED", message: "All workstreams failed. Fix failures before integrating." },
    };
  }

  // Load plan and verify (optional, warn)
  const planArtifact = await loadPlanArtifact(repoRoot);
  if (!planArtifact) {
    console.warn("Warning: plan.json not found. Proceeding without plan context.");
  }

  const verifyArtifact = await loadVerifyArtifact(repoRoot);
  if (!verifyArtifact) {
    console.warn("Warning: verify.json not found. Proceeding without verify context.");
  }

  // Build the AI prompt
  console.log("Building integration test prompt...");
  const { prompt, promptHash, detectedFramework } = await buildIntegrationTestPrompt({
    executeArtifact,
    planArtifact,
    verifyArtifact,
    repoRoot,
    testFramework: options.testFramework,
  });

  // Call the AI model
  console.log(`[AI] Calling model for integration test generation...`);
  console.log(`[AI] Detected framework: ${detectedFramework.name} (${detectedFramework.language})`);

  let aiModelUsed = "";
  let aiConfig: any;
  let rawResponse = "";

  try {
    const result = await executeWorkstream(prompt, repoRoot);
    aiModelUsed = result.modelUsed;
    aiConfig = result.aiConfig;
    rawResponse = result.response ?? "";
  } catch (err) {
    const message = err instanceof Error ? err.message : String(err);
    return {
      status: "failed",
      summary: `AI model call failed: ${message}`,
      artifactPath: "",
      outputRoot: repoRoot,
      failure: { code: "AI_GENERATION_FAILED", message },
    };
  }

  // Parse AI response — extract JSON array
  let generatedFiles: Array<{ path: string; framework: string; language: string; content: string }> = [];
  
  try {
    // Try to extract JSON array from response
    const jsonMatch = rawResponse.match(/\[[\s\S]*\]/);
    if (!jsonMatch) throw new Error("No JSON array found in AI response");
    
    const parsed = JSON.parse(jsonMatch[0]);
    if (!Array.isArray(parsed)) throw new Error("AI response was not an array");
    
    generatedFiles = parsed.map((item: any, i: number) => ({
      path: item.path ?? `tests/integration/generated-test-${i + 1}.test.${detectedFramework.language === "python" ? "py" : "ts"}`,
      framework: item.framework ?? detectedFramework.name,
      language: item.language ?? detectedFramework.language,
      content: item.content ?? "",
    }));
  } catch (err) {
    const message = err instanceof Error ? err.message : String(err);
    return {
      status: "failed",
      summary: `Failed to parse AI response: ${message}`,
      artifactPath: "",
      outputRoot: repoRoot,
      failure: { code: "AI_GENERATION_FAILED", message: `Failed to parse AI response: ${message}` },
    };
  }

  if (generatedFiles.length === 0) {
    return {
      status: "failed",
      summary: "AI generated no test files.",
      artifactPath: "",
      outputRoot: repoRoot,
      failure: { code: "NO_TEST_FILES_GENERATED", message: "AI returned empty test files array." },
    };
  }

  console.log(`[AI] Generated ${generatedFiles.length} test file(s).`);

  // Write test files and count tests in each
  const testFiles: IntegrationTestFile[] = generatedFiles.map(tf => ({
    path: tf.path,
    framework: tf.framework,
    language: tf.language,
    testCount: countTests(tf.content, tf.framework),
    content: tf.content,
  }));

  // Run the tests
  console.log(`[Tests] Running integration tests: ${detectedFramework.testCommand}`);
  const testResult = await runIntegrationTests(
    testFiles,
    repoRoot,
    options.testFramework ?? detectedFramework.testCommand
  );

  // Build artifact
  const artifact = buildIntegrateArtifact(
    executeArtifact,
    planArtifact,
    verifyArtifact,
    testResult,
    aiModelUsed,
    aiConfig,
    SCHEMA_VERSION,
    FORGE_VERSION,
  );

  // Write outputs
  const outputDir = options.outputDir ?? path.join(repoRoot, ".forge");
  const artifactPath = path.join(outputDir, "integrate.json");
  const reportPath = path.join(outputDir, "integration-report.md");

  try {
    await writeIntegrateArtifact(artifactPath, artifact);
    await fs.writeFile(reportPath, createIntegrationReport(artifact), "utf-8");
  } catch (err) {
    const message = err instanceof Error ? err.message : String(err);
    return {
      status: "failed",
      summary: `Failed to write outputs: ${message}`,
      artifactPath,
      reportPath,
      outputRoot: outputDir,
      failure: { code: "IO_ERROR", message },
    };
  }

  // Summary output
  const { summary } = artifact;
  const statusIcon = summary.failed > 0 ? "✗" : "✓";
  console.log(`\n${statusIcon} Integration complete.`);
  console.log(`  Tests: ${summary.passed} passed, ${summary.failed} failed, ${summary.skipped} skipped`);
  console.log(`  Duration: ${summary.durationMs}ms`);
  console.log(`  Test files: ${summary.testFilesGenerated}`);
  console.log(`\n  Artifact: ${artifactPath}`);
  console.log(`  Report: ${reportPath}`);

  return {
    status: summary.failed > 0 ? "failed" : "ready",
    summary: `Integration: ${summary.passed}/${summary.total} tests passed`,
    artifactPath,
    reportPath,
    outputRoot: outputDir,
    exitCode: summary.failed > 0 ? 1 : 0,
  };
}

function countTests(content: string, framework: string): number {
  if (framework === "pytest") {
    return (content.match(/def test_\w+/g) ?? []).length;
  }
  return (content.match(/\bit\s*\(/g) ?? []).length;
}
```

```typescript
import type { IntegrateArtifact } from "./types.js";

export function createIntegrationReport(artifact: IntegrateArtifact): string {
  const { summary, tests, testFiles, workstreamsSummary, goal, recommendations } = artifact;
  
  const lines: string[] = [
    "# Forge Integration Report",
    "",
    `**Date:** ${artifact.createdAt}`,
    `**Goal:** ${goal}`,
    "",
    "## Workstreams Summary",
    "",
    `| Total | Completed | Failed | Changes Made |`,
    `|-------|-----------|--------|-------------|`,
    `| ${workstreamsSummary.total} | ${workstreamsSummary.completed} | ${workstreamsSummary.failed} | ${workstreamsSummary.totalChangesMade} |`,
    "",
    "## Integration Test Results",
    "",
    `**AI Model:** ${summary.aiModelUsed ?? "unknown"}`,
    "",
    `| Result | Count |`,
    `|--------|-------|`,
    `| ✅ Passed | ${summary.passed} |`,
    `| ❌ Failed | ${summary.failed} |`,
    `| ⏭️ Skipped | ${summary.skipped} |`,
    `| ⏱️ Duration | ${summary.durationMs}ms |`,
    "",
    "## Test Files Generated",
    "",
  ];

  if (testFiles.length === 0) {
    lines.push("*No test files were generated.*");
  } else {
    for (const tf of testFiles) {
      lines.push(`- **${tf.path}** — ${tf.testCount} test(s) (${tf.framework}, ${tf.language})`);
    }
  }

  lines.push("");
  lines.push("## Individual Test Results");
  lines.push("");

  if (tests.length === 0) {
    lines.push("*No tests were run.*");
  } else {
    for (const test of tests) {
      const icon = test.status === "passed" ? "✅" : test.status === "failed" ? "❌" : "⏭️";
      lines.push(`### ${icon} ${test.name}`);
      if (test.error) {
        lines.push(`\`\`\`\n${test.error}\n\`\`\``);
      }
      if (test.recommendation && test.status === "failed") {
        lines.push(`**Recommendation:** ${test.recommendation}`);
      }
      lines.push("");
    }
  }

  if (recommendations.length > 0) {
    lines.push("## AI Recommendations for Fixing Failures");
    lines.push("");
    for (const rec of recommendations) {
      lines.push(`- ${rec}`);
    }
    lines.push("");
  }

  // Next steps
  lines.push("## Next Steps");
  lines.push("");
  if (summary.failed === 0) {
    lines.push("✅ **All tests passed.** Integration is complete. The codebase is ready.");
  } else {
    lines.push(`❌ **${summary.failed} test(s) failed.** Review the failures above and fix the issues.`);
    lines.push("");
    lines.push("Suggested workflow:");
    lines.push("1. Review failed test errors and recommendations above");
    lines.push("2. Fix the identified issues in the relevant files");
    lines.push("3. Re-run `forge integrate` to verify fixes");
    lines.push("4. If tests still fail, manually inspect the generated test files");
  }

  return lines.join("\n");
}
```

```typescript
export function formatIntegrateCommandOutput(result: IntegrateCommandResult): string {
  const lines = [
    `Status: ${result.status}`,
    `Summary: ${result.summary}`,
    result.outputRoot ? `Output root: ${result.outputRoot}` : null,
    result.artifactPath ? `Artifact: ${result.artifactPath}` : null,
```
